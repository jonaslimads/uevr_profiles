cmake_minimum_required(VERSION 3.20)
project(SWGRCutsceneProbe LANGUAGES CXX)
add_library(SWGRCutsceneProbe SHARED Probe.cpp)
target_compile_features(SWGRCutsceneProbe PRIVATE cxx_std_20)
set(UEVR_SOURCE_DIR "" CACHE PATH "Path to a UEVR source checkout")
if(NOT EXISTS "${UEVR_SOURCE_DIR}/include/uevr/API.h")
    message(FATAL_ERROR "Set UEVR_SOURCE_DIR to a UEVR checkout containing include/uevr/API.h")
endif()
target_include_directories(SWGRCutsceneProbe PRIVATE "${UEVR_SOURCE_DIR}/include")
target_link_libraries(SWGRCutsceneProbe PRIVATE user32)
target_compile_options(SWGRCutsceneProbe PRIVATE /W4 /WX)
